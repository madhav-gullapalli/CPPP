CP++ Correctness Documentation
==============================

This file documents correctly written CP++ programs and snippets that should
compile or run successfully.

Declarations
------------

  int declarations compile.

  Example:
    int x = 3;

primitive float declaration
  float declarations compile.

  Example:
    float x = 2.5;

primitive char declaration
  char declarations compile.

  Example:
    char c = 'x';

  Example:

  Example:

  List<int> declarations compile.

Functions
---------

function definition and call
  CP++ supports top-level user-defined functions with typed parameters and
  typed returns.

  Example:
    int square(int x){
        return x * x;
    }
    print(square(5));

  Output:
    25

vector alias parameter
  Function parameters may use `vector<T>` as an alias for `List<T>`.

  Example:
    int sumV(vector<int> values){
        return sum(values);
    }
    List<int> values = [1, 2, 3, 4];
    print(sumV(values));

  Output:
    10

void recursive function
  `void` functions may use `return;`, may recurse, and List/string parameters
  are passed by reference in generated C++ so recursive graph-style helpers
  can mutate caller-owned state.

  Example:
    void dfs(List<List<int>> adj, List<bool> vis, int u){
        vis[u] = true;
        print(u, end = " ");
        for (int v in adj[u]) {
            if (!vis[v]) {
                dfs(adj, vis, v);
            }
        }
    }
    List<List<int>> adj = [[1, 2], [0, 3], [0], [1]];
    List<bool> vis = [false, false, false, false];
    dfs(adj, vis, 0);

  Output:
    0 1 3 2

Synthesis tests
---------------

tree dfs traversal
  Build an undirected tree from `n` and `n - 1` edges, sort every adjacency
  list, then run a recursive DFS from node 0.

  Example:
    void dfs(List<List<int>> adj, List<bool> vis, int u){
        vis[u] = true;
        print(u, end = " ");
        for (int v in adj[u]) {
            if (!vis[v]) {
                dfs(adj, vis, v);
            }
        }
    }
    int n = input();
    List<List<int>> edges = input(n - 1, 2);
    List<List<int>> adj = [];
    rep(n) {
        adj.add([]);
    }
    for (int i = 0; i < len(edges); i++) {
        int a = edges[i][0];
        int b = edges[i][1];
        adj[a].add(b);
        adj[b].add(a);
    }
    for (int i = 0; i < n; i++) {
        adj[i].sort();
    }
    List<bool> vis = [];
    rep(n) {
        vis.add(false);
    }
    dfs(adj, vis, 0);
    print();

  Input:
    4
    0 1
    0 2
    1 3

  Output:
    0 1 3 2

  Input:
    5
    0 4
    4 2
    0 1
    1 3

  Output:
    0 1 3 4 2

array rotation by k
  Rotate a list right by k positions.

  Example:
    int n = input();
    int k = input();
    List<int> values = input();
    List<int> rotated = [];
    int shift = k % n;
    for (int i = 0; i < n; i++) {
        int from = (i - shift + n) % n;
        rotated.add(values[from]);
    }
    print(rotated);

  Input:
    5
    2
    1 2 3 4 5

  Output:
    [4, 5, 1, 2, 3]

  Input:
    4
    5
    10 20 30 40

  Output:
    [40, 10, 20, 30]

longest increasing contiguous segment
  Find the length of the longest strictly increasing contiguous run.

    int n = input();

            current++;
        } else {
            current = 1;

partition around x
  Partition a list into < x, == x, and > x while preserving order.

  Example:
    int n = input();
    int x = input();
    List<int> values = input();
    List<int> out = [];
    for (int i = 0; i < n; i++) {
        if (values[i] < x) {
            out.add(values[i]);
        }
    }
    for (int j = 0; j < n; j++) {
        if (values[j] == x) {
            out.add(values[j]);
        }
    }
    for (int k = 0; k < n; k++) {
        if (values[k] > x) {
            out.add(values[k]);
        }
    }
    print(out);

  Input:
    7
    3
    3 1 4 3 2 5 3

  Output:
    [1, 2, 3, 3, 3, 4, 5]

  Input:
    5
    10
    9 10 11 10 8

  Output:
    [9, 8, 10, 10, 11]

array is almost sorted
  Check whether removing at most one element can make the list strictly increasing.

  Example:
    List<int> values = input();
    bool ok = false;
    for (int skip = -1; skip < len(values); skip++) {
        bool good = true;
        bool started = false;
        int prev = 0;
        for (int i = 0; i < len(values); i++) {
            if (i == skip) {
                continue;
            }
            if (!started) {
                prev = values[i];
                started = true;
            } else {
                if (values[i] <= prev) {
                    good = false;
                }
                prev = values[i];
            }
        }
        if (good) {
            ok = true;
        }
    }
    print(ok);

  Input:
    1 2 5 3 4

  Output:
    1

  Input:
    3 2 1

  Output:
    0
